//program to find out palindrome or not
#include <stdio.h>
#include <conio.h>

int main()
{
    int num, original, digit, reverse = 0;
clrscr();

    printf("Enter a number: ");
    scanf("%d", &num);

    original = num;

    while (num > 0)
    {
        digit = num % 10;
        reverse = reverse * 10 + digit;
        num = num / 10;
    }

    if (original == reverse)
        printf("Palindrome");
    else
        printf("Not Palindrome");

    getch();
}
